cmake_minimum_required(VERSION 3.16)
project(railbridge C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

add_executable(railbridge_test tests/test_core.c src/core.c src/parse.c src/rumble.c)
target_include_directories(railbridge_test PRIVATE include src)
target_compile_definitions(railbridge_test PRIVATE RB_NO_USB)

find_package(PkgConfig QUIET)
set(HAVE_USB OFF)
if (PkgConfig_FOUND)
  pkg_check_modules(LIBUSB IMPORTED_TARGET libusb-1.0)
  if (LIBUSB_FOUND)
    set(HAVE_USB ON)
  endif()
endif()

if (HAVE_USB)
  add_executable(railbridge src/core.c src/parse.c src/rumble.c src/host.c src/gadget.c src/net.c src/main.c)
  target_include_directories(railbridge PRIVATE include src)
  target_link_libraries(railbridge PRIVATE PkgConfig::LIBUSB)
else()
  add_executable(railbridge src/core.c src/parse.c src/rumble.c src/gadget.c src/net.c src/main.c)
  target_include_directories(railbridge PRIVATE include src)
  target_compile_definitions(railbridge PRIVATE RB_NO_USB)
endif()

if (WIN32)
  target_link_libraries(railbridge PRIVATE ws2_32)
else()
  find_package(Threads REQUIRED)
  target_link_libraries(railbridge PRIVATE Threads::Threads)
  target_link_libraries(railbridge_test PRIVATE Threads::Threads)
endif()

enable_testing()
add_test(NAME core COMMAND railbridge_test)
